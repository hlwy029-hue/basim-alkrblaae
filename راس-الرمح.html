كل منحرا الشايله جاوب يا راس الرمح

الشاعر: أحمد العبادي | أداء: الحاج باسم الكربلائي

‎كـــــــل مـــنــحــر الــشــايــلـه             چم  لــــوعــــة  وشچم  جــــــــرح ‎أشـــتــعــرض لـــهـــل عــايــلــه             جـــــــاوب  يــــــراس الـــرمـــح ‎يـــــــراس  أصـــــــد يــــــراس             لـــبـــن أمـــــي لـــــو عـــبــاس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎ ‎** ‎ ‎ويــلـي أمـــن أصـــد ويــا الــدرب             ونـــــــــــــار الـــــقــــلــــب ‎تــــتـــلاهـــب أبـــحـــســراتــي ‎ ‎يــتـحـدى صــبـري كـــل صــعـب             تـــــطــــلــــع غــــــصــــــب ‎مــــاتــــنــــدف آهــــــاتــــــي ‎ ‎ ‎أنـــــــــــظـــــــــــر  وچم راس             يـــــــلــــــوح لـــعـــيـــونـــي ‎وأنــــــــــــدب الــــعــــبـــاس             كــــــافــــــل أضــــعــــونــــي ‎وعــالــخــد تـــصـــب دمــعــاتـي ‎ ‎ ‎بــــيـــن  أخــــوتـــي أمــدلــلــه             مــهــيـوبـه عـــفـــت الـــوطـــن ‎لأنــــــــي ســـبــيــه أبـــفــلــه             ويـــــــلاه يــــغـــدر الـــزمـــن ‎وأســــمـــع  شـــتـــم أرجــــــاس             لـــبـــن أمـــــي لـــــو عـــبــاس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎ ‎ ‎ ‎ ‎رد  يـــــرمــــح وأذكـــــرأمــــس             ضـــــــــــــي الــــشــــمــــس ‎وحــســيــن   أخـــويـــه  أقــبــالـي ‎ ‎لــــو قــلـتـك عــــودك مـايـحـس             مــــــــــــــن ألـــــتــــمــــس ‎وأشـــــــرح  هـــضـــم دلالــــــي ‎ ‎ ‎دم نــــــحــــــر عــــــبــــــاس             مـــــــن   خــــضـــب  راســـــــه ‎لابــــــــــــد الــــنــــومــــاس             حــــــــــــرك      أحــــســــاســـه ‎وأدري بــــنـــحـــر كـــفـــالـــي ‎ ‎ ‎بــــــراس  الأخــــــو الــحــامـلـه             بــغــيـرة وجـــبــل مـــــن وفــــى ‎وظــــعـــن الــــبـــدر كــافــلــه             حــــــــي وأبـــــــد مـــاغــفــى ‎يــــلـــوق لــــمـــن الــنــومــاس             لـــبـــن أمـــــي لـــــو عـــبــاس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎ ‎ ‎ ‎ ‎ ‎قـــــــر يــــرمـــح بــالـمـظـلـمـة             بـــــــهــــــا لـــمـــحـــكــمــة ‎نــــدري الــذنــب مــــو ذنــبــك ‎ ‎دم الــســفــك أشـــــرف دمـــــى             أرض وســـــــــــــمـــــــــــــى ‎وبــهــالــنــحـور أمـــخــضــبــك ‎ ‎ ‎أحــــــفــــــظ   ألــــشــــالـــك             وأشــــهـــد أعــــلـــى الـــصـــار ‎وســــــولـــــف  أهــــــوالـــــك             لــــلـــنـــبـــي الـــمـــخـــتـــار ‎وأنـــقـــل مــصــايــب دربـــــك ‎ ‎ ‎أروي الـــقـــصـــص كـــامـــلــة             وأعــرضــهــا يــــــوم الــحــشــر ‎ســــولـــف لـــعــقــد الـــولـــة             وجـــدامـــة  شــنــهــو  الـــعـــذر ‎وشـــتـــعـــتــذر  هــــالـــنـــاس             لـــبـــن أمـــــي لـــــو عـــبــاس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎ ‎ ‎ ‎ ‎ ‎جــــــاوب   يــــــروح  الــنــايـبـة             وشـــــــتـــــــجـــــــاوبــــــة ‎لــيــنـظـر  خـــضــاب أبـــراســك ‎ ‎مـــــن يــنــحـر طــبـعـه الابــــة             ومـــــــــــــن   خـــــضــــبــــه ‎وأعــــــرف   ذنـــــب  حـــراســك ‎ ‎ ‎تـــــــدري نــــحـــر حـــســيــن             يـــــنــــفــــس  شـــــــمــــــة ‎وتـــــــــــدري بــالــكــونــيــن             مـــــنــــزلــــت  أمـــــــــــــة ‎ومــنــهــن  أخــــــذ  مــقـيـاسـك ‎ ‎ ‎نــــحـــرة الـــنــبــي الــقــبــلـة             شـــتــريــد أعــــــدد شــــــرف ‎راســــــــه الــــذكـــر رتــــلـــه             وشــفــافــة تـــقـــرى الــكــهـف ‎دوم الـــــشـــــرف يـــنـــقـــاس             لـــبـــن أمـــــي لـــــو عـــبــاس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎ ‎ ‎ ‎ ‎صــــد يــرمــح جــمــر الــغـضـى             مـــــــــــــن تـــــعــــرضــــى ‎ابــــــدرب  الــيــســر جـــدامــي ‎ ‎وأنــشـدنـي عــــن لــيــل الـمـضـى             شــــــلــــــون أنــــقــــضــــى ‎وأنـــشــدنــي عــــــن أيــتــامــي ‎ ‎ ‎لازلـــــــــــــم  ضــــــلــــــت             تــــــحـــــرس الــــصــــيـــوان ‎وبـــالـــخـــيـــم لــــهــــبــــت             يـــــالــــرمــــح نــــــيــــــران ‎بـــــــس الــــحـــرم بــخــيــامـي ‎ ‎ ‎أشـــروانـــي طــــــف كــربــلــه             مــــاظـــل لادمــــــع لاصـــبـــر ‎ويـــــــا مــــوقـــف أتــحــمـلـه             وشــــــوف الـــرزايـــا أشــكــثـر ‎أمن أنظر طفل ينداس لبن أمي لو عباس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎ ‎ ‎ ‎ ‎شـــــحچي  وشــسـولـف يــارمــح             عـــــــــــــن   يـــــــاجــــــرح ‎عــــن  يــاهـضـم  بــيــش  أبــديـه ‎ ‎صـــعــدت شــمــوسـك يـــارمــح             ويــــــــــــــا الـــــصــــبــــح ‎كــلــهـن قــطــع مــــن چبــــدي ‎ ‎ ‎مــــــــن تــــصــــد عـــيــنــي             الــــــــــــرووس ولــــيــــانـــي ‎والله تـــــــــاذيـــــــــنــــــــي             لـــــــوعــــــة  أحــــــزانــــــي ‎وغـــيـــر الـــدمـــع مــاعــنــدي ‎ ‎ ‎بـــــــس دمـــعــتــي الــهــامـلـة             تـــلــهــب  جـــمـــر  عــالــوجــن ‎حــــســـراتـــي  مـــتـــواصــلــة             بـــيــهــا  الــقــلــب مــنــتـحـن ‎وشـــيــحــمــل    الأحـــــســــاس             لـــبـــن أمـــــي لـــــو عـــبــاس ‎جـــــــاوب  يــــــراس الـــرمـــح ‎
