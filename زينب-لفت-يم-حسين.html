زينب لفت يم حسين

الشاعر: عبدالحسين الشرع | الرادود: باسم الكربلائي
زيــنــب لــفــت يــم حــســيـن ... لـﭽن ﮔابـعـه بـالـهـم
تـﮔلــه يــا ضــوه عـيـونـي ... عـلـيـمـن هـالـفـزع مـلـتـم
🔹🔹🔹🔹🔹
تـعـنـت لـيـه لـلـخـيـمـه ... وتـفـسر الصخر ونتهـا
طـبـت قـعـدت اﮔبـالـه ... وعـالـخـد تـهـل دمـعتهـا
تـﮔلـه اعـلـيـك ضـلـع امـك ... الـمـظـلـومـه اومـصـيـبـتـها
سـولـف لـي يـمـاي الـعـين ... لا تخفي علي يحسيــن ... عليمن هالفزع صوبين
وأشــوف ابــكـثـر عـج الـخـيـل ... وادي كـربـلـه غـيـم
🔹🔹🔹🔹🔹
أويـلـي مـن سـمـعـهـا احـسـيـن ... سـالـت دمـعة اعيونـه
يـﮔلــهــا أخـاف أسـولـف لـﭻ ... او وجـهـﭻ يـنـخـطـف لـونـه
ابـﭽتــلـى او ﭽتـل أهـل بـيـتـي ... يـخـتـي الـﮔوم يـردونـه
ولا بـد مـا تـشـوفـيـنـه ... فوﮔ الثره امـﮔطعينه ... يزينب لا تنوحينــه
عـيـنـج عـالـيـتامه النـار ... لو شبت بالمخيـم
🔹🔹🔹🔹🔹
تـﮔلــه الـكـاتـبـك يـحـسـيــن ... مـن هـالـنـاس ﭽـا ويـنـه
يـﮔل لـهـا يـمـاي الـعـيـن ... كـلـهـا انـﮔلـبـت اعـلـيـنـه
تـﮔل لــه حــمــل اضــعــونـك ... او سـدر لـلـوطـن بـيـنـه
يـﮔل لـهـا مـا يـخـلوني ... أسدر بعد بضعوني ... نيتهم يـﭽتلونـي
ولابـد مـا تـشـوفـيـنـي ... او شـيـبـي يـﮔطـر امـن الدم
🔹🔹🔹🔹🔹
والأكــبــر عــالأرض مـطـروح ... يـخـتـي عـيـونـج اتـشـوفـه
او جــاسـم عـالـثـرى مـرمـي ... ابـدال الـعـرس والـحـوفـه
وأخـوج اعـلـى الـنـهـر نـايـم ... ويـمـه امـﮔطـعـه اﭽفـوفـه
او يـمـه رايـتـه او جـوده ... او عـيـنـه ابسهم ممروده ... بطل حيلي على ازنوده
لـو شـفـتـيـهـن ابكتـره ... او جسمه اعلى الثره امخذم
🔹🔹🔹🔹🔹
يـخـتـي او سـهـم الـمـثـلـث ... ابـدلالـي تـشـوفـيـنـه
او عــبــدالله الــطـفـل يـخـتـي ... ابـدم نـحـره امـطـوقـيـنـه
او لــو شــفــتـي الـشـمـر يـمـي ... ابـعـدي او لا تـحـاﭽيـنـه
تره هو يـﮔطع ابنحـري ... او يتربع على صـدري ... يزينب عايني او صبري
لــمــن يــحــكــم الــبــاري ... او هــوه ابـحـالـنـا أرحـم
