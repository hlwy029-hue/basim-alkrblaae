حضر عندك علي جدك

الشاعر: جابر الكاظمي | الرادود: باسم الكربلائي

عــلــي جــدك ... حــضـر عـنـدك
سـمـيـك يا علي الأكبر ... ينوح المصرعك حيدر
يـشـمـك مـن دمـه الـمـنحر ... علي جدك

يـا الأكـبـر بـالـحـرب لـمـن طحت مجروح
حـضـر عـنـدك عـلي و شافك جسد مطروح
انـحـنـه و يـنـظـر جـسم دامي و نحر مذبوح
گعـد يـم الـجـسـد يـحـسـب طعن و جروح
شـهگ شـهگت حـزن و يـصـب دمع مسفوح
عـجـيـبـة الـلـي فـتـح خـيبر يون و ينوح
الــجـفـن يـنـبـوع ... يـصـب ادمـوع
يــصـب ادمـوع ... الـجـفـن يـنـبـوع
گعــد يــمــك ... يـشـم جـسـمـك
يــشــم جــسـمـك ... گعـد يـمـك
يـصـد لـجروحك بعبره ... يعد جم طعنة جم طبره
جـبـدتـه تـسـتـعـر جـمـره ... مثل جبدك
🔹🔹🔹🔹🔹
سـمـيـك يـا عـلـي الأكـبـر يـبـث اهموم
مـثـل حـوم الگطـا الـتـايـه عـلـيك ايحوم
گعـد يـم جـسـمـك بـحـسرة و گلب مالوم
صـدگ هـذا الـوصـي حـيـدر دمـعته اتموم
عـجـيـبـة فـارس الـخـنـدگ يـطل مهموم
عـجـيـبـة الـلـي جـتـل مرحب ينوح اليوم
عـجـيـبـة ايـلـيـن ... بـطـل صـفـيـن
بـطـل صـفـيـن ... عـجـيـبـة ايـلـيـن
الــجــتـل مـرحـب ... گعـد يـنـحـب
گعــد يــنـحـب ... الـجـتـل مـرحـب
تـفـطـر بـالهضم جبده ... بدت هالحالة من عنده
يـجـر حـسـرة ووضـع خـده ... عـلى خدك
🔹🔹🔹🔹🔹
عـجـيـبـة الـلـي يـدب نفسه على الكلفات
نـشـوفـه الـلـيـلـة يـا الأكـبر يهل دمعات
عـجـيـبـة الـلـي فـنـى و دمر هبل و اللآت
يـجـر ونّـه بـأثـر ونّـه و يـجـر حـسـرات
عـجـيـبـة الـبـحـرب اسـمـه ابو الحملات
يـنـوح و مـن جـفـن عـيـنـه يصب عبرات
حـلـيـف الـسـيـف ... يـون يـا حـيـف
يـون يـا حـيـف ... حـلـيـف الـسـيـف
يــصـب دمـعـه ... انـعـطـف ضـلـعـه
انــعـطـف ضـلـعـه ... يـصـب دمـعـه
لـون عن حالته اتناشد ... وگف وگفت ابو الفاجد
يـصـد بـعـيـونـه و يـشـاهد ... ذبل وردك

يـصـد لـك يـا عـلـي الأكـبـر علي الكرار
يگلـك يـا بـنـي مـا تـدري اشـجرالي و صار
لـفـيـت و مـن شـفـت حـالك صرت محتار
ونـيـنـك هـيّـج اجـروحـي و سـلب أفكار
أخـمّـد جـمـرت اهـمـومـي بـدمـع فـجّار
و انـشّـف دمـعـة اعـيـونـي بـلـهيب النار
يــا الأكــبـر جـيـت ... الـك ونـيـت
الــك ونــيـت ... يـا الأكـبـر جـيـت
هــضــم دهــري ... سـلـب فـكـري
ســلــب فــكــري... هـضـم دهـري
و انا الـلـي شاعت اخباري ... يهز الدنيا بتّاري
اجـيـتـك و الـدمـع جـاري ... الهضم فگدك

يـا الأكـبـر يـا بـنـي مـن شفتك على التربان
احـس بـضـامـري اتـوجـد سـنـه الـنيران
الـعـذب مـلـك امـك الـزهره و مهرها جان
تــشـربـه و تـرتـوي الـوادم بـلا أثـمـان
و ابـنـهـا انـت يـا ولـيـدي مـتت عطشان
و اشـوفـك يـا بـنـي بـالـمـذبـح بلا أجفان
تـحـن خلـوك ... و لا رووك
و لا رووك ... تـحـن خلـوك
بــقــلــب ضـامـي ... مـتـت دامـي
مــتــت دامــي ... بگلــب ضــامـي
اون الـحـالـتك و انحب ... اشوفك بالدمه امخضب
يـشـب و يـنـطـفـي و يـلهب ... جمر فگدك

يـا الأكـبـر يـا نـبـع روحـي و نمير احشاي
مـتـت تـتـحـسـر بـلـهفة اعلى شربة ماي
عـلـي يـا بـنـي و صـرت لـجلك شريد الراي
و انا الـمـا بـارح اعـيـونـي كـدر دنـياي
اجـيـت و طـبـرتـي بـراسـي جـبـتها وياي
يـا الأكـبـر و شـكـثـر تـشـبه طبرتك هاي
عـلـى الـجـرحـيـن ... بـجـت كـل عـين
بـجـت كـل عـيـن ... عـلـى الـجـرحـين
بــجــمــر هــمــي ... نــزف دمـي
نــزف دمــي ... بــجــمــر هــمـي
نـزف دمي مثل دمك ... وقع جسمي مثل جسمك
عـلـي يـا بـنـي يـطـل يمك ... علي بلحدك

و انـزلـك و اسـكـن بـلـحـدك يا ضي العين

مــسـيـل ادمـاك ... خـضـبـنـي ويـاك
