سيد الرسل يفتح سجل

الشاعر: جابر الكاظمي | الرادود: باسم الكربلائي
سـيـد الرسل.. يـفـتـح سجل.. و لأمته يعاتبها
مـيـظـل عـذر.. يـوم الـحـشر ..و الخالق يحاسبها
و عــلـيـهـا تـشـهـد ....عـتـرة مـحـمـد
دم الـجره يصير شهود.. يم الإله المعبود ..و الخالق يحاسبها
🔹🔹🔹🔹🔹
مـن يـبـدي الـحـشـر يـومـه و يـصير حساب
رســول الله يـجـي و شـايـل بـجـفـه كـتـاب
و يـبـدي الـنـبـي عـلـى الامـة يـوجـه عتاب
عـلـى كـل دم جـرى و كـل نـايـبـة و مصاب
أول عـتـب.. شـنـهـو السبب ..ضلع الوديعة تكسَّر
لـيـش انـطبر ..فارس مُضَر.. شنهو الصدر ..من حيدر
يــسـأل خـصـيـمـه.. عـن كـل جـريـمـه
جـتل الحسن سره يعود ..يم الإله المعبود..و الخالق يحاسبها
🔹🔹🔹🔹🔹
و عــلــرابــع مـصـيـبـة سـيـد الأكـوان
سـؤالـه يـوجـه لـشـيـعـة بـنـي سـفـيـان
لـيـش بـكـربـلاء جـهـزتـوا هـالـفـرسـان
و لـيـش حـسـيـن صـار لـخـيـلـهـم ميدان
ابـنـي نـغـدر ..دمـه انـهـدر ..عالغبرة ليش اتقطع
و بـيـا عـدل.. دمـه يـحل.. منكم جواب ارد اسمع
مـنـهـو الـلـي صـابـه.. سـبـب مـصـابـه
دم ابـنـي لسا موجود ..يم الإله المعبود.. و الخالق يحاسبها
🔹🔹🔹🔹🔹
دم ابـنـي انـسـفـك مـن جـسـمـه و تـخـذم
الــيــوم قـبـال ربـنـا يـنـعـرض هـالـدم
و الــنـار الـحـرقـتـوا بـيـهـا الـمـخـيـم
هــي الـلـي تـدخـلـكـم عـلـى جـهـنـم
حـرقـتـوا الخيم ..فوق الحَرَم.. حل يا ذنب من عندها
بـديـرة غـرب.. خـيل و رعب.. لا والي يحرس حدها
راحــت سبيــة.. وي ال أمــيــة
أصحاب الوجوه السود.. يم الإله المعبود ..و الخالق يحاسبها
🔹🔹🔹🔹🔹
يــا شــيــعـة بـنـي سـفـيـان يـا اشـرار
هـذي عـيـلـتـي و آنـة الـنـبـي الـمـخـتار
و شــلـون بـسـبـي تـمـشـوهـا عـلأكـوار
و انــتــو تــدَّعــون الإسـلام يـا فـجـار
و بـيـا سـنـن ..تمشي بضعن.. بنات الوحي و تتيسّر
شـنـهو الـبـده.. منها شسده.. تتطلع سبية الخِدَر
الــخِــدَّر غــرايــب ..بــيـن الأجـانـب
كل حق على الأمة يعود.. يم الإله المعبود ..و الخالق يحاسبها
