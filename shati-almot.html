يا شاطي الموت حاجيني

الشاعر: جابر الكاظمي | سنة الإصدار: 2005
ملاحظة: تبدئ القصيدة في الدقيقة 1:21

يـا شاطــي المـوت حاجــــيني انـا الــــعبـاس تدريــني اريد الــمــاي... اريد الــمــاي تـريـد انــطـيـك چفـيـني اخـذهـا وهـاك ضـي عـيـني وأبيـع دمــاي... اريد الــمــاي __________________________________________________________ نـهـر الفـرات ويـاك جيـت احچي واتـمنـاك تــسمــعنــي لــو نــاديــت عيـالـي تـركـتـها هنـاك عـطـشانـه تتـرجـاك هــذا إنــي يــمــك جــيــت... هــذا إنــي يــمــك جــيــت مـا تـدري مـنـهـو الـجـاك عـبـاس إنـا الـفتـاك والـجـرفـك تعـنـيـت إي والـذي إجـراك بـالسـيـف اسـد مجـراك لــو بـالــعــذب شــحــيــت... لــو بالــعــذب شــحــيــت عـفـت اطـفـال بـخـيـامـي اروي الـيـوم كل ضـامـي إنـا الـعبـاس... إنـا الـعبـاس قـصـدت والـمـوت جـدامـي سـحكت الجـيـش بـجـدامـي عـلـى رغـم عـداي... إريد الماي __________________________________________________________ بـالـخـيـمـه عـفـت عـيـال عـطشـانـه والإطـــفـــال تــشـــرب نــمــيــر الـعـــيـــن وإنـا عـلـى ذاك الـحـال بـمـنـاثـر الـدلـال ودعــت إخـــويـــه حــســيــن... ودعــت إخـــويـــه حــســيــن وبـمـدمع الـهـمـال ويـايـه يـحچي وكـال عــلــيــك اخــشــى مــن البــيــن يـابـالـفضـل اهـوال خـيـل وفـزع ورجـال تتـــحاوشــك صــوبــين... تتـــحاوشــك صــوبــين صـحت يـا حـسـيـن لايـهـمك الـعـلامـه ويـاي ضـلـع امـك إنـا الـعـبـاس... إنـا الـعبـاس الـخـيـار يـصـيـر مـن خـصـمك تـريـد الـمـاي لـو دمـك إكـول بـهـاي... إريـد الـمـاي __________________________________________________________ ودعـتـه للـمـضـلـوم جـيـت وشـفـت هـالـكـوم جــدامــي مــثــل الســيــل شـفـت الـدرب مـلـزوم خـيـل ورجـال تـحـوم تتــخـطــى عـالــيــاويــل... تتــخـطــى عـالــيــاويــل عـلـيـهـا الإجـل مـرسـوم عـطـشـى وتـريـد دمـوم وصـاعـــد عــجـــاج الــخــيــل والـحـومـه چـنـهـا غـيـوم واصـبـح نـهـاري الـيـوم يــشــبــه ســواد اللـــيــل... يــشــبــه ســواد اللـــيــل رجـال وخـيـل حـاطـتنـي صـحـت والـكـوم سـمـعتنـي إنـا الـعبـاس... إنـا الـعبـاس ثـدايـه الـطـيـب رضـعتنـي وحجـور الـزود ربـتنـي وأنـا الـوفـاي... إريـد الـمـاي __________________________________________________________ عـزمـي إنـفـجـر بـركـان وانـذهـلـت الـفـرسـان مــا تــدري ويـــــن تــروح وتـمـاوج المـيـدان والـدم صـبـح طـوفـان مــهــري ســفــينــة نـــوح... مــهــري ســفــينــة نـــوح طـايـف عـلـى الإبـدان واتـغـطـت الـتـربـان بمــوج الــدمــى الـــمســـفــوح شـردت بـلا اذهـان مـن شـافـت الـعـدوان درب الـــمــفـر مــفتــوح... درب الـــمــفـر مــفتــوح الـخـصـم بالـسـيـف ذلـيـتـه عـلـى درب الـمـوت دلـيـتـه إنـا الـعبـاس... إنـا الـعبـاس دمـاه يـفـوج خـلـيـته سـواد اللـيـل راويـته گـصـدت عـالـنـاي... إريـد الـمـاي __________________________________________________________ چـان الـدرب مـسـدود خـيـل وزلـم وحـشـود مــا هــمــنــي وتــخطـــيت ويـايـه جـبـت الـجـود هـذا إنـا بـحـر الـزود يــامــا زلــم ذلــيــت... يــامــا زلــم ذلــيــت يـالـشـاطـي وأنـت شـهـود خـل يـشـهـد الـمعـبـود بــهـذا الـــعهـــد وفـــيـــت حـفـظ الـعهـد مـريـود لازم أرد ردود تــدري شـــگلت شــحجــيت... تــدري شـــگلت شــحجــيت إذا هــالـمـاي مـاحـصـلـه الـنـبع أفـنـيـه مـن إصـلـه إنـا الـعبـاس... إنـا الـعبـاس إرد بالـجـود يـا مـحـلـه إضـحـي الـروح مـن إجـلـه ويـضـل رجـواي... إريـد الـمـاي __________________________________________________________ هـنـا يـالـفـرات تـشـوف بـالـمـاي إرد مـلـهـوف وأتـــعنـــى للأيـــتــام إرجـع وإنـا الـمـعـروف مـا يـهمـنـي جـيـش وخـوف مــاتــهــمنــي هــالإزلــام... مــاتــهــمنــي هــالإزلــام وشـمـا تـجـيـنـي صـفـوف رمـاح ونـبـل وسـيـوف إفــنــيــهــا بــالــصـمـصــام لـو گـطـعـولـي چـفـوف بـالـجـود إرد زحـوف وإگــصــد إهــل وخـــيــام... وأگــصــد إهــل وخـــيــام حـرم واطـفـال يـتنـونـي إجـيـب الـمـاي يـردوني إنـا الـعبـاس... إنـا الـعبـاس غـدَ الإشـرار يـلـفـونـي بـسـهـم بـالـعـيـن يـرمـونـي گـطـع يـمـنـاي... إريـد الـمـاي يـا شـــاطـي المــوت حاجـــيني إنـا الـعبـاس تـدريـني إريـد الـمـاي... إريـد الـمـاي تـريـد انـطـيـك چـفـيني إخـذهـا وهـاك ضــي عـيـنـي وإبـيع دمـاي... إريـد الـمـاي
