صلى عليك مليك السماء

الشاعر: جابر الكاظمي | أداء: الحاج باسم الكربلائي

صــلــى عــلــيــك مــلـيـكُ الـسـمـاء جـسـمٌ صـريـعٌ فـي الـعـراء .... مـقـطعُ الاعضاء سـلـيـب .... حـسـيـن.... غـريـب .... حـسـين مـو آيـة وحـده بهذي نزلت آيات .... بيها الكتاب يصرِّح شـمـا بـالـسما و الأرض من مخلوقات .... بسم الإله تسبِّح مـعـنـاهـا يـشعر عالم الموجودات .... باليحزن و باليفرح و بـهالشعور عليك ناحت كل ذات .... وادي الطفوف يوضّح بــكــى عــلــيــك ضــمـيـرُ الـجـمـاد ثـم ارتـدى ثـوبَ الـحـداد .... يـا مُـبـكـي الأطواد عـجـيـب .... حـسـيـن.... غـريـب .... حـسين أول صـواب الـحـجـر لـمن صابك .... ذاك الجبين اتهشّم هـذا الحجر ظل ينحب على مصابك .... و أصبح جبينك مأتم سـهـم الـمـثـلـث سِـجَد يم أعتابك أدى التحية و سلَّم و مـن صـوب چبـدك ظل يحن لصوابك .... ألّمك و هو تألم شـابـت عـلـيـك رؤوس الـنـبـال و بـكـتـك صـالـيـة الـقـتـال .... من محجرٍ هطّال حـبـيـب .... حـسـيـن.... غـريـب .... حـسـين لـمـن طِـحِت و استقبلت وجه الله .... بدم الجروح الترجف تـعـنـاك شـمر الخنا و سيفه بيمناه .... لقطع النحر متلهف و عالنحر من وسّط السيف و خلاّه .... سيف الشمر ظل يرجف هالسيف ظل يبچي على نحرك و دماه .... و على الجراح التنزف بــكــت عــلــيــك عــيـونٌ الـسـيـوف و تـراب بـاديـة الـطـفـوف .... فـي خـطبك المعروف تـريـب.... حـسـيـن.... غـريـب .... حـسـيـن عـن جثتك يا حسين راسك فصلوه .... جسمك صريع مترَّب تـراب الأرض يبچي على حالك خلوه .... بموجة دماك تخضّب و الـشـمس من شافت وريدك قطعوه .... بالحزن ظلت تلهب راسـك بـراس الـرمح لمن شالوه .... حتى الرمح ظل ينحب جـرت عـلـيـك دمـوعُ الـرمـاح و بـكـتـك بـارقـة الـصـفـاح .... يا من إلى الأرواح مـجـيـب .... حـسـيـن.... غـريـب .... حـسـين والـدك يـوم الـحـشر هو الميزان .... و بچفه حوض الكوثر شـلون انته ابنه تموت بالطف عطشان .... چبدك ذِبَل و اتفطر نـهـر الـفـرات الـيوم منك خجلان .... من چبدتك يتعذر خـلـوك يـمـه بـلا غُسِل بلا أچفان .... مغسَّل بدم المنحر نــاحــت عــلــيــك ضــفــاف الـفـرات دمـعٌ و مـوجٌ جـاريـات .... و أنـتَ فـي الـفـلوات خـصـيـب.... حـسـيـن.... غـريـب .... حـسين حـتـه الـخـيـم حزنانه يبن الأطهار .... حيرتك و تنظرها هـاي الـخـيـم مـا وجروها الأشرار .... جمر الحزن وجرها مـن كـثـر ما حزنة إلك شبت نار .... اعليك القهر سجرها و عـين السحاب اعليك سالت مدرار .... ناعي الأجل كدرها هــلــت عــلــيــكَ جـفـونُ الـسـحـاب و بـكـت لـمـصـرعـكَ الـهـضاب .... و أنت للألباب قـريـب.... حـسـيـن.... غـريـب .... حـسـيـن ابكل عين إلك دمعة حزن يا مظلوم .... ابكل قلب لهفه و حسره مـو بـس جـماد الأرض ينعاك اليوم .... ينحب الكون بأسره يـحـسـيـن وشـما بالسموات انجوم .... أجرت عليك العبره حـزنـك على اوجوه الكواكب مرسوم .... يبن البتول الزهره هـوت عـلـيـكَ شـمـوسُ البهاء .... بالحزنُ أثقلها العزاء إذ أنـت لـلـعـلـيـاء ربـيـب .... حـسـيـن .... غـريـب .... حـسـيـن
