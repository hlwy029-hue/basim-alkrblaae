گبـل مـاتـروح و تـهـاجـر ....يـطير الصاف تانيني اخـذنـي ويـاك لأحـبـابـي ...و على الجنحان خليني و ادوِّر ويـن أبـويـة حـسين....و أدوِّر وين ابوية حسين ************ گبـل مـا يـطـوِّح الـحـادي قـبل ما ياخذ الأحباب تـرجـيـتـه بـدمـع يـطـفح مثل ما يطفح الخنياب يـهـداك الله يـا حـاديـهـم تـهدّى و لا تسوق الناب مـشـى و لا جـلـمة من عنده سمعت و لا سمعت جواب جزى و بكل مسرى كل خطوة أحس تركت جرح و صواب تـبـاريـهـم غـدت عـيـني و تصد لما شبحهم غاب يـا طـيـر الـصـاف عـافـونـي و آنـة الياويل يبرالي أصـد لـلـبـيـت بـعـيوني و أشوف البيت ظل خالي قــطــع لا ويــن..... ابــويــة حـسـيـن ************ عـسـاهـا تـهـدمـت داري عـلية و ريت موتي حان و عـسـى بـروح الأجـل يـنعى و لا اسمع حد الركبان أحـاجـي الـدار و اسـألـهـا بـعد هم ترجع الوليان سـكـوت و لا جـواب أسـمع مظلمة و موحشة الجدران عـلـى فـراش الـمـرض جـسمي و أرامح رمحة الغرقان و أتـرجّـى جـرف صـبـري لـعـلـه يخفف الأحزان عـسـاه الـمـوت يـلـفـيـنـي يا طير الصاف و أتمنى و لا دنـيـاي اقـضـيـهـا عـلـى الآهـات و الونة قــصــد لــلـبـيـن.... ابـويـة حـسـيـن ************* صِـحِـت ولـيـاي عـافـونـي دواهـي البين حاطتني مـثـل حـال انـولـد مـذبـوح يا طير الصاف خلتني عَـلَـي تـتـغـايـر هـمـومـي غصص دنياي ذهلتني غـصـص مـا يـنـحـمل همها على نعش الموت شالتني عـلـيـلـة و مـن زغـر سـنـي نـحيلة و المرض فتني يـا طـيـر الـصـاف و الـدنـيا اشكثر لوعات راوتني عـلـى الـلـوعـات أتـسـنَّد و أطيح ردود من همي و اديـر الـعـيـن و أتـلـفّـت أبـويـة حسين مو يمي تــرك جــرحــيــن.... ابــويـة حـسـيـن *********** جـرح مـن لـوعـة فـراقـه انـحُـفر من الدمع خدي جـرح ثـانـي جـرح قـتّـال مـن سـيل المرض يبدي عـفـانـي شـلـون هـالـجرحين ما قطعن هِرِش جبدي عـلـى هـالـحالة يا طير الصاف جزيت من الصبر حدي سـواد الـلـيـل يـلـفـيـني و اقولن يقضي و يعدي و أصـيـح اشـمـطـولك يا ليل و لا نسمة صبح تبدي أوِن و الـلـيـل مـايـرحـم ونـيـنـي و سيل عبراتي و آنـة الـمـلـيـت مـن هـمي و نحيبي و نار حسراتي أجــيــب مــنــيـن ....أبـويـة حـسـيـن *********** حـجـيـت آنـة ويـا حـاديـهم قبل ما بالظعن يطلع الـحـادي مـا بـدى لـي جـواب ترك موح الجفن ينبع حـجـيـت الـنـوب ويـا الـدار محاجي و ياها ماتنفع حـجـيـت ويـا الـنـجم و الليل ما منهن جواب اسمع حـجـيـت ويـا الـنـسيم الهاب جواب النسِم مايرجع عـفـتـهـم كـلـهـن و جيتك يا طير الصاف بالمدمع أريـد جـواب مـن عـنـدك حـلـيـفة نوح تدريني أمـشّـي عـلـيـك بـاريـنـا يا طير الصاف حاجيني تــريــد الــعــيــن ....ابـويـة حـسـيـن *********** لَـو انـك رايـد تـهـاجـر تـوانـي ويـاي و اتـمهَّل أشـد روحـي عـلـى جنحانك و اخذني وياك من ترحل نـروح و نـقـطـع الـبـيـدا ندوِّر على هلي و نسأل و نـتـخـطـى عـلـى ظهر الغيم عسى المايوس يتحًّصل يـا طـيـر الصاف و اريد اوصيك لو انك شاهدت محمل بـرخـى عـلـى الـقـاع نـزِّلني القلب ماعاد يتحمل حـمـلـت هـمـوم لـو صـبَّـت على الجلمود يتفطَّر غـيـاب حـسـيـن شـب قلبي و صليل حشاي يتوجر شــعــل نــاريــن.... أبــويـة حـسـيـن *********** صـدق نـرحـل يـا طـير الصاف و على جناحك تخليني و صـدق نـوصـل لـعـد أهـلي و اشوفن الوالد بعيني لـبـيـابـي صـبـيـهـن اهدي لك لون لأهلي توديني و الـك ثـلـثـيـن من عمري يا مقصر عمري و سنيني ريـاحـيـن الـعـمـر ذبـلـت و لا خـفقة برياحيني و ابـاري مـنـاحـر مـصـابـي ثـجـيل الهم يباريني لـظـى الـحـسـرات هـيّمني هِمِت و الروح مفجوعة بـسـقـم بـلـواي مـحـتـارة حِـرِت مابين هاللوعة حِــرِت مــابــيــن.... أبــويـة حـسـيـن
