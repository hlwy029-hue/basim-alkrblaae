مثل طبع النسر

الشاعر: الأديب جابر الكاظمي | أداء: الحاج باسم الكربلائي

مـثـل طـبـع الـنـسر طبعي اطير بهيبة الجنحين واحـس يـتـكسر اجناحي من اذكر مصاب حسين حســـــــــينأً واحُسيــــــــنا حـجب غيم الكدر شمسي وعلي وجهي الصباح اظلم ابـات وحـسـرتـي بصدري الجبل من عدها يتهدم و سـحـابة عيني على الوجنة غدت تمطر دموع ودم حـزيـن وصـبـري يـعصيني ويطاوعني مرار البين حســـــــــينأً واحُسيــــــــنا يـوصـفـولـي الـصـبر وانا ادور عالصبر ماصاح تـهـدم سـاتـر جفوني ودمعتي اعلى الوجن تسفح عـلـى الـي اخـوتـه وقـومه وبينينه بخطته المذبح يـشـاهـدـم جثث صرعه ضحايه اتناثرو صوبين حســـــــــينأً واحُسيــــــــنا تـفـرق شـمـلـه واتـوزع بـين الشفرة والمنحر وبـيـن امـواج خـيل اعداه يضل وحده ابو الاكبر تـتـعـاصـف ريـاح الـموت ويتلافح هجير الحر وحـيـد اسـتـوحـدوه الكوم حاير يتجه لا وين حســـــــــينأً واحُسيــــــــنا غـيـر الـسـيـف مـاعنده مناصر بالوغى ينصره رمـوه بـسـهـم الـمـثلث وتكور من ضهر مهره وكـع وكـعـه اخـوه اللي وكع من عصرو الزهره مـااشـبـه امـس بـالـيوم وجمعة كربله مصابين حســـــــــينأً واحُسيــــــــنا وكـع ضـامـي وجـمـر قلبه انتقل من قلبه لخيامه بـحـوافـر خـيـلهم سحكَوه مثل ماسحكَوا ايتامه رمـاح اعـلـى الـجسد واسهام من كل كتر تترامه وهـجـت خـدر وايـتـام مـاتـعرف دربها منين حســـــــــينأً واحُسيــــــــنا انـفـصـل عـن الجسم راسه وكع طود الصبر يمه يـتـامـه وخـدر انـسـلبت مثل ماسلبوا جسمه انـطـفـح دم مـنحره وحتى المنايه اختنكَت بدمه بـكـل عـيـن انرسم نحره ودمه اتفايض بكل عين حســـــــــينأً واحُسيــــــــنا
