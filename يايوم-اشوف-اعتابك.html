يا يوم أشوف أعتابك

الشاعر: الأديب جابر الكاظمي | أداء: الحاج باسم الكربلائي

يـا يـوم أشـوف أعـتـابـك - وأوگف حـزيـن بـبـابـك وأرجــع أزورك يــا حـــيــــــدر طـول عـلـيـه غـيـابـك - مـشـتـاگ أشـم تـرابـك وأرجــع أزورك يــا حـــيــــــدر آنــه إلــهــويــتــك - لا تــظــن نـسـيـتـك عـلـي وهذا الإسم يسري بشرياني - نبع يل ما تمل شوگك رياحيني عـتـب ويـاك عـندي يراوح بعيني - يمن فرگاك كدر صفوة سنيني أريـد أحچي بـصـراحة وياك خليني - أخاف مكدرات الجو تنسيني يـا بـا الـحـسـن مـا تـدري - صـرت أشـتـريـها بعمري شـوفـت ضـريـحـك يـــا حـيـدر مـن الـفـرگة نـافـذ صـبـري - وظـلـت تـعـيش بفكري شـوفـت ضـريـحـك يـــا حـيـدر بــروحــي إشــتــريـتـك - لا تـظـن نـسـيـتـك عـشـت أعوام يمك يا ربيع الحي - ما أعوفك چنت مثل الجرف والمي لـو صرت بهجير أگصد إلك يا في - ولو ليل الوكت دلهم أشوفك ظي غـصـب من جنتك طلعوني أهل الغي - وخلوني أظل لا ميت ولا حي مــن جـنـتـك حـرمـونـي - بـرجـوة أمـل خـلـونـي كـل فــكـري يـمـك يــا حــيــدر عـنـك غـصـب بـعـدونـي - أمـشـي وأديـر عـيـونـي كـل فــكـري يـمـك يــا حــيــدر عــشــت بــثــنـيـتـك - لا تـظـن نـسـيـتـك مو بيدي غصب عفتك يا بو الحسنين - وعنك رحت أتلفت وأدير العين گلـت هالفرگة بالميت شهر شهرين - وطالت يا علي لا سنة ولا سنتين أنـاشـد عالصبر ما أدري دربه منين - خذن ثلثين حيلي مخلفات البين دنـيـايـه نـفـرت عـنـي - وحـتـى الـصـبـر مـل مـني مـشـتـاگ أشــوفــك يــا حـيـدر أغـفـى بـوسـادة ظـنـي - وأصـحـى عـلـى دمـعي ووني مـشـتـاگ أشــوفــك يــا حـيـدر يــا مــا نــخــيــتــك - لا تـظـن نـسـيـتـك البعد شيبني يا من فرگتك كلفة - وجرح شوگي غميج وما وگف نزفه لـيـالـي الظيم باگن بسمة الشفة - وسنيني إنگضت بالحسرة واللهفة لچن بـعده الأمل بالروح ما يخفه - وأحس روحي بعدها لهالوكت ترفه مـن عـيـنـي يـنـزف مـوحـي - وآنـه بـتـرافـة روحي أرســم خـيـالـك يــا حــيــدر هـالـحـبـر دم جـروحـي - وصـرت بـصـحـيـيفة نوحي أرســم خـيـالـك يــا حــيــدر بــروحــي لگيــتــك - لا تــظــن نـسـيـتـك بـخـيالك عايش من سنين وأتحسر - ولا أدري مصيري شلون يا حيدر صـرت مـا أشعر بطعم البرد والحر - وإذا نسمة هوى تزاجمني أتكدر جـمـر ظـيـمي بصميم حشاي يتوجر - وبعيني غدت عبراتي تتكسر مـا عـنـدي غـفـوة بـنـومـي - ولا صـحـوة تجلي همومي حـيـرنــي بـعـدك يــا حــيــدر مـا مـش سـعـد بـنـجـومـي - ولا فـرحـة تـسعد يومي حـيـرنــي بـعـدك يــا حــيــدر تــرحــمــنــي ريــتــك - لا تـظـن نـسـيـتـك يا وهجة عشگ ناغت خلايا الدم - خلافك عمري ذرة وضايعة بمنجم بـغـيرك شمل روحي ما بعد يلتم - يا بو الحسنين إنت الجرح والبلسم يا يوم أوصلك وأكشف سحاب الهم - وأنفض يم ضريحك كل غبار الهم يـا يـوم أصـل يـم گبـرك - وأنـصـب مـنـاحـة بـكترك و أبچي لـمـصـابـك يــا حــيــدر يــل مــا صـفـى لـك دهـرك - أذكـر مـرارة صـبـرك و أبچي لـمـصـابـك يــا حــيــدر وأذكــر رزيــتــك - لا تــظــن نــســيــتـك أتـمـنـى مـثل هاليوم چنت هناك - وأجدد يم ضريحك هالعهد وياك وأنـصب مأتم على مصابك وأنعاك - وأون ون الثكالة لگقدك وفرگاك لـكن ما نسيتك عايش بذكراك - وضميري من البعد يا با الحسن ناداك بـيـن الـدمـع والـحـسـرة - أذكـر لـهـيـب الـطـبـرة يـصـعـب مـغـيـبـك يــا حـيـدر مـن عـيـنـي تـجـري الـعـبـرة - وبـيـن الـجـوانح جمرة يـصـعـب مـغـيـبـك يــا حـيـدر بــالــهــم بچيــتــك - لا تــظــن نـسـيـتـك
